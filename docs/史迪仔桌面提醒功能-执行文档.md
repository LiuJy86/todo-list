## 一、功能概述

当待办事项到达设定时间时，屏幕上弹出**史迪仔（Stitch）角色动画**，配合抖动效果 + 文字气泡提醒用户，实现趣味性强、视觉冲击力大的桌面提醒。

### 核心体验

```
┌─────────────────────────────────────────────────────────┐
│                                                         │
│    ╭──────────────╮                                     │
│    │  ⏰ 该开会了！ │  ← 文字气泡（带动画弹出）           │
│    ╰──────┬───────╯                                     │
│           │                                             │
│      ╭────┴────╮                                        │
│      │  (^_^)  │  ← 史迪仔 GIF（抖动动画）               │
│      │  史迪仔  │                                        │
│      ╰─────────╯                                        │
│                                                         │
│   [知道了] [稍后提醒(5分钟)] [完成待办]                    │
│                                                         │
└─────────────────────────────────────────────────────────┘
```

#### 3.1.2 创建提醒窗口函数

```javascript
/**
 * 创建史迪仔提醒窗口
 * @param {Object} todo - 待办事项数据 { id, title, notes, priority }
 */
function createReminderWindow(todo) {
  // 如果提醒窗口已存在，先关闭
  if (reminderWindow) {
    reminderWindow.destroy();
    reminderWindow = null;
  }

  const { screen } = require('electron');
  const display = screen.getPrimaryDisplay();
  const { width: screenW, height: screenH } = display.workArea;

  // 创建透明、无边框、置顶窗口
  reminderWindow = new BrowserWindow({
    width: 320,
    height: 380,
    x: screenW - 340,           // 右侧留 20px 边距
    y: screenH - 400,           // 底部留 20px 边距
    frame: false,               // 无边框
    transparent: true,          // 透明背景
    alwaysOnTop: true,          // 始终置顶
    skipTaskbar: true,          // 不在任务栏显示
    resizable: false,           // 不可调整大小
    focusable: true,            // 可获取焦点
    hasShadow: false,           // 无阴影（透明窗口）
    webPreferences: {
      contextIsolation: true,
      nodeIntegration: false,
      preload: path.join(__dirname, 'preload.js')
    }
  });

  // 加载提醒页面，传递待办数据
  const todoData = encodeURIComponent(JSON.stringify(todo));
  reminderWindow.loadFile(
    path.join(__dirname, '..', 'src', 'reminder.html'),
    { query: { todo: todoData } }
  );

  // 窗口关闭时清理引用
  reminderWindow.on('closed', () => {
    reminderWindow = null;
  });

  // 窗口显示后播放动画（通过 IPC 通知渲染进程）
  reminderWindow.webContents.on('did-finish-load', () => {
    // 可选：播放提示音
    // 由渲染进程处理，避免主进程阻塞
  });
}
```

#### 3.1.3 定时器管理函数

```javascript
/**
 * 为待办事项设置提醒定时器
 * @param {Object} todo - 待办事项 { id, title, reminderTime, ... }
 */
function setReminderTimer(todo) {
  // 清除该待办已有的定时器
  clearReminderTimer(todo.id);

  if (!todo.reminderTime) return;

  const now = Date.now();
  const triggerTime = new Date(todo.reminderTime).getTime();
  const delay = triggerTime - now;

  // 如果时间已过，不设置定时器
  if (delay <= 0) return;

  // 设置定时器（最大 24.8 天，超过需用 setInterval）
  const timerId = setTimeout(() => {
    // 时间到！创建提醒窗口
    createReminderWindow(todo);
    // 从 Map 中移除
    reminderTimers.delete(todo.id);
    // 通知主窗口刷新状态
    if (mainWindow) {
      mainWindow.webContents.send('reminder-triggered', todo.id);
    }
  }, delay);

  // 保存引用
  reminderTimers.set(todo.id, timerId);

  console.log(`[提醒] "${todo.title}" 将在 ${new Date(triggerTime).toLocaleString()} 提醒`);
}

/**
 * 清除待办事项的提醒定时器
 * @param {string} todoId - 待办事项 ID
 */
function clearReminderTimer(todoId) {
  const timerId = reminderTimers.get(todoId);
  if (timerId) {
    clearTimeout(timerId);
    reminderTimers.delete(todoId);
    console.log(`[提醒] 已取消待办 ${todoId} 的提醒`);
  }
}

/**
 * 清除所有提醒定时器
 */
function clearAllReminders() {
  reminderTimers.forEach((timerId) => clearTimeout(timerId));
  reminderTimers.clear();
  console.log('[提醒] 已清除所有提醒定时器');
}
```

#### 3.1.4 IPC 处理

```javascript
// 在 ipcMain 区域新增

// 设置提醒（渲染进程 → 主进程）
ipcMain.on('set-reminder', function (_, todo) {
  setReminderTimer(todo);
});

// 取消提醒
ipcMain.on('cancel-reminder', function (_, todoId) {
  clearReminderTimer(todoId);
});

// 稍后提醒（延迟 N 分钟）
ipcMain.on('snooze-reminder', function (_, todo, minutes) {
  // 先清除原有定时器
  clearReminderTimer(todo.id);
  // 设置新的触发时间
  const newTodo = { ...todo };
  newTodo.reminderTime = new Date(Date.now() + minutes * 60000).toISOString();
  setReminderTimer(newTodo);
});

// 关闭提醒窗口
ipcMain.on('close-reminder', function () {
  if (reminderWindow) {
    reminderWindow.destroy();
    reminderWindow = null;
  }
});

// 完成待办（从提醒窗口触发）
ipcMain.on('complete-todo-from-reminder', function (_, todoId) {
  // 通知主窗口标记完成
  if (mainWindow) {
    mainWindow.webContents.send('complete-todo', todoId);
  }
  // 关闭提醒窗口
  if (reminderWindow) {
    reminderWindow.destroy();
    reminderWindow = null;
  }
});
```

### 3.5 提醒窗口脚本 (`src/js/reminder.js`)

```javascript
/**
 * 史迪仔提醒窗口脚本
 * 负责：解析待办数据、控制动画、处理用户交互
 */

(function () {
  'use strict';

  // ============================================
  // 获取 URL 参数中的待办数据
  // ============================================
  function getTodoFromURL() {
    const params = new URLSearchParams(window.location.search);
    const todoEncoded = params.get('todo');
    if (!todoEncoded) return null;
    try {
      return JSON.parse(decodeURIComponent(todoEncoded));
    } catch (e) {
      console.error('解析待办数据失败:', e);
      return null;
    }
  }

  // ============================================
  // 初始化
  // ============================================
  function init() {
    const todo = getTodoFromURL();
    if (!todo) {
      console.error('未获取到待办数据');
      return;
    }

    // 填充数据
    populateData(todo);

    // 启动动画
    startAnimation();

    // 播放提示音
    playSound();

    // 绑定按钮事件
    bindEvents(todo);
  }

  // ============================================
  // 填充页面数据
  // ============================================
  function populateData(todo) {
    const reminderText = document.getElementById('reminder-text');
    const todoTitle = document.getElementById('todo-title');
    const todoNotes = document.getElementById('todo-notes');
    const stitchImg = document.getElementById('stitch-img');

    // 设置提醒文字
    reminderText.textContent = getReminderText(todo);

    // 设置待办标题
    todoTitle.textContent = todo.title || '待办事项';

    // 设置备注（如果有）
    if (todo.notes) {
      todoNotes.textContent = todo.notes;
      todoNotes.style.display = 'block';
    } else {
      todoNotes.style.display = 'none';
    }

    // 根据优先级选择不同史迪仔表情
    const gifIndex = getStitchGifByPriority(todo.priority);
    stitchImg.src = `img/史迪奇${gifIndex}.gif`;
  }

  // ============================================
  // 根据场景生成提醒文字
  // ============================================
  function getReminderText(todo) {
    const texts = [
      `该${todo.title}了！`,
      `别忘了：${todo.title}`,
      `⏰ ${todo.title} 时间到！`,
      `嘿！该${todo.title}啦～`,
    ];
    // 根据 todo.id 选择一个固定的文案（保证同一待办每次显示一致）
    const index = hashString(todo.id) % texts.length;
    return texts[index];
  }

  // ============================================
  // 根据优先级选择史迪仔 GIF
  // ============================================
  function getStitchGifByPriority(priority) {
    // 1=普通(可爱), 2=中等(期待), 3=紧急(着急)
    switch (priority) {
      case 3: return 6;  // 紧急 - 史迪奇6.gif（夸张表情）
      case 2: return 4;  // 中等 - 史迪奇4.gif
      case 1:
      default: return 1; // 普通 - 史迪奇1.gif（可爱）
    }
  }

  // ============================================
  // 启动动画序列
  // ============================================
  function startAnimation() {
    const stitchWrapper = document.getElementById('stitch-wrapper');

    // 延迟启动抖动，让入场动画先播放
    setTimeout(() => {
      stitchWrapper.classList.add('shaking');
    }, 800);

    // 10 秒后自动停止抖动（避免一直抖）
    setTimeout(() => {
      stitchWrapper.classList.remove('shaking');
    }, 10000);
  }

  // ============================================
  // 播放提示音
  // ============================================
  function playSound() {
    try {
      const audio = new Audio('提示音效.mp3');
      audio.volume = 0.6;
      audio.play().catch(e => console.log('音频播放失败:', e));
    } catch (e) {
      console.log('音频播放失败:', e);
    }
  }

  // ============================================
  // 绑定按钮事件
  // ============================================
  function bindEvents(todo) {
    const container = document.getElementById('reminder-container');

    // "知道了"按钮 - 关闭窗口
    document.getElementById('btn-dismiss').addEventListener('click', () => {
      dismissWithAnimation();
    });

    // "稍后提醒"按钮 - 延迟 5 分钟
    document.getElementById('btn-snooze').addEventListener('click', () => {
      if (window.electronAPI) {
        window.electronAPI.snoozeReminder(todo, 5);
      }
      dismissWithAnimation();
    });

    // "完成"按钮 - 标记待办完成
    document.getElementById('btn-complete').addEventListener('click', () => {
      if (window.electronAPI) {
        window.electronAPI.completeTodoFromReminder(todo.id);
      }
      dismissWithAnimation();
    });

    // 点击史迪仔也可以关闭
    document.getElementById('stitch-wrapper').addEventListener('click', () => {
      dismissWithAnimation();
    });
  }

  // ============================================
  // 带动画关闭
  // ============================================
  function dismissWithAnimation() {
    const container = document.getElementById('reminder-container');
    container.classList.add('dismissing');

    // 动画结束后通知主进程关闭窗口
    setTimeout(() => {
      if (window.electronAPI) {
        window.electronAPI.closeReminder();
      }
    }, 300);
  }

  // ============================================
  // 工具函数：字符串哈希
  // ============================================
  function hashString(str) {
    let hash = 0;
    for (let i = 0; i < str.length; i++) {
      const char = str.charCodeAt(i);
      hash = ((hash << 5) - hash) + char;
      hash = hash & hash; // Convert to 32bit integer
    }
    return Math.abs(hash);
  }

  // ============================================
  // 启动
  // ============================================
  document.addEventListener('DOMContentLoaded', init);

})();
```

### 3.6 待办管理扩展

在现有的待办管理代码中，添加提醒触发逻辑：

```javascript
/**
 * 当用户保存待办时，同步设置提醒定时器
 * 在 todo.js 的 saveTodo 函数中调用
 */
function onTodoSaved(todo) {
  // 如果待办有提醒时间，通知主进程设置定时器
  if (todo.reminderTime && window.electronAPI) {
    window.electronAPI.setReminder(todo);
  }
}

/**
 * 当用户删除待办时，同步取消提醒
 */
function onTodoDeleted(todoId) {
  if (window.electronAPI) {
    window.electronAPI.cancelReminder(todoId);
  }
}

/**
 * 当用户完成待办时，同步取消提醒
 */
function onTodoCompleted(todoId) {
  if (window.electronAPI) {
    window.electronAPI.cancelReminder(todoId);
  }
}

/**
 * 应用启动时，恢复所有待办的提醒定时器
 */
function restoreAllReminders() {
  const todos = loadTodos(); // 加载所有待办
  todos.forEach(todo => {
    if (!todo.completed && todo.reminderTime) {
      const triggerTime = new Date(todo.reminderTime).getTime();
      if (triggerTime > Date.now()) {
        // 未过期，设置定时器
        window.electronAPI.setReminder(todo);
      }
    }
  });
}
```

## 七、风险与注意事项

### 7.1 技术风险

| 风险 | 影响 | 应对措施 |
|------|------|----------|
| 透明窗口在某些显卡驱动下显示异常 | 黑屏或白屏 | 提供设置项允许关闭透明效果 |
| setTimeout 最大延迟限制（24.8 天） | 长期待办无法触发 | 超过 24 小时的用 setInterval 轮询 |
| 多显示器环境下窗口位置异常 | 提醒显示在屏幕外 | 使用 `screen.getCursorDisplay()` 获取当前屏幕 |
| 电脑休眠期间定时器暂停 | 错过提醒 | 监听 `resume` 事件，触发过期提醒 |

### 7.2 用户体验注意事项

- 抖动动画**最多持续 10 秒**，避免一直抖造成干扰
- 提醒窗口**不抢焦点**（`focusable: false` 可选），避免打断工作流
- 提供**全局开关**，允许用户关闭史迪仔提醒
- 提供**勿扰模式**（如全屏游戏时不弹出）
